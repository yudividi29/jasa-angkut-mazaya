// --- Backend Jasa Angkut Mazaya (Versi Pro V.5 - Full Control) --- function doGet(e) { if (e.parameter.action === 'getData') { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheetPesanan = ss.getSheetByName('Pesanan'); var sheetDriver = ss.getSheetByName('Driver'); var sheetPelanggan = ss.getSheetByName('Pelanggan'); var dataPesanan = sheetPesanan ? sheetPesanan.getDataRange().getDisplayValues() : []; var dataDriver = sheetDriver ? sheetDriver.getDataRange().getDisplayValues() : []; var dataPelanggan = sheetPelanggan ? sheetPelanggan.getDataRange().getDisplayValues() : []; return ContentService.createTextOutput(JSON.stringify({ status: 'success', pesanan: dataPesanan, driver: dataDriver, pelanggan: dataPelanggan })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput("Sistem Database Jasa Angkut Mazaya ONLINE."); } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var action = data.action; if (action === 'saveOrder') return saveOrderData(data.payload); else if (action === 'addDriver') return addSheetData('Driver', [data.payload.id, data.payload.nama, data.payload.posisi, data.payload.hp, data.payload.plat, 'Aktif']); else if (action === 'updateOrder') return updateOrderData(data.payload); // FITUR BARU return createJsonResponse({ status: 'error', message: 'Action tidak dikenal.' }); } catch (error) { return createJsonResponse({ status: 'error', message: error.toString() }); } } // FUNGSI BARU: UPDATE PESANAN LANGSUNG DARI DASHBOARD function updateOrderData(payload) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('Pesanan'); if (!sheet) return createJsonResponse({ status: 'error', message: 'Sheet tidak ditemukan' }); var data = sheet.getDataRange().getValues(); var orderId = payload.orderId; // Cari baris yang sesuai dengan ID Pesanan for (var i = 1; i < data.length; i++) { if (data[i][0] === orderId) { var rowNum = i + 1; // Baris keberapa di excel sheet.getRange(rowNum, 14).setValue(payload.statusOrder); // Kolom 14: Status Perjalanan sheet.getRange(rowNum, 15).setValue(payload.statusBayar); // Kolom 15: Pembayaran sheet.getRange(rowNum, 16).setValue(payload.sopir); // Kolom 16: Sopir sheet.getRange(rowNum, 17).setValue(payload.kernet); // Kolom 17: Kernet sheet.getRange(rowNum, 18).setValue(payload.armada); // Kolom 18: Armada return createJsonResponse({ status: 'success' }); } } return createJsonResponse({ status: 'error', message: 'Order tidak ditemukan' }); } function saveOrderData(payload) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheetPesanan = ss.getSheetByName('Pesanan'); var date = new Date(); var dateStr = Utilities.formatDate(date, "GMT+7", "yyyyMMdd"); var jamStr = Utilities.formatDate(date, "GMT+7", "dd/MM/yyyy HH:mm:ss"); var noOrder = "ORD-" + dateStr + "-" + Math.floor(1000 + Math.random() * 9000); var dimensiStr = (payload.panjang || 0) + "x" + (payload.lebar || 0) + "x" + (payload.tinggi || 0) + " cm"; var rowPesanan = [ noOrder, jamStr, payload.nama || "", payload.whatsapp || "", payload.alamatJemput || "", payload.alamatTujuan || "", payload.jarak || "", payload.estimasiWaktu || "", payload.berat || 0, dimensiStr, payload.keterangan || "-", payload.fotoStatus || "-", payload.grandTotal || 0, 'MENUNGGU', 'BELUM LUNAS', '-', '-', '-' ]; sheetPesanan.appendRow(rowPesanan); var sheetPelanggan = ss.getSheetByName('Pelanggan'); if (sheetPelanggan) { var dbPelanggan = sheetPelanggan.getDataRange().getValues(); var sudahAda = false; for (var i = 1; i < dbPelanggan.length; i++) { if (dbPelanggan[i][2] == payload.whatsapp) { sudahAda = true; break; } } if (!sudahAda) sheetPelanggan.appendRow(["CUST-" + Math.floor(10000+Math.random()*90000), payload.nama, payload.whatsapp, payload.alamatJemput, jamStr]); } return createJsonResponse({ status: 'success', noOrder: noOrder }); } function addSheetData(sheetName, rowData) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(sheetName); if (sheet) sheet.appendRow(rowData); return createJsonResponse({ status: 'success' }); } function createJsonResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON); }
